# Mailbox ECCSA — dependencias del backend.
#
# These versions come from the ECOSA ecosystem mandate and live in
# ECCSA-Shell/versiones/requisitos-canonicos.txt — it's Field's list. Field is
# the reference and is not modified; Mailbox matches it on everything the apps
# share, so a PDF or a push notification looks and behaves the same no matter
# which app sends it.
#
# What's NOT here (and why):
#   reportlab / Pillow → Mailbox doesn't generate PDFs, and it doesn't PROCESS
#                        images either: a signature image is validated by its
#                        magic bytes and its dimensions are read straight from the
#                        PNG/GIF/JPEG/WEBP header (see `_es_imagen` and
#                        `_dimensiones` in api/main.py). Pillow would have been
#                        ~40 MB for two integers that only set the logo's layout.
#                        A previous version imported PIL to do exactly that and
#                        every upload failed with a 500 plus a file orphaned in the
#                        volume, because the import was outside the `try`.
#   google-generativeai → no IA in this app.
#   pysmb → the SMB share is used by the WORKER, in another container. This app
#            doesn't touch the share.
#   selenium / Playwright → not used.

# ── API web ───────────────────────────────────────────────────────────────────

uvicorn[standard]==0.30.0

# ── Autenticación ─────────────────────────────────────────────────────────────
# pyjwt: the stateless challenges of the passkeys (WebAuthn) are JWTs, so they
#        survive a restart and work with more than one worker.
# webauthn: verification of passkeys, with rpId at the ecc-sa.com.mx root.
pyjwt==2.9.0
webauthn==3.0.0

# ── Notificaciones push ───────────────────────────────────────────────────────
pywebpush==2.0.0
py-vapid==1.9.1
# cryptography NO se fija. `webauthn==3.0.0` (las passkeys, el mismo par que usa
# Admon) exige `cryptography>=49`, así que fijarla en 43 hace que `pip install`
# falle con ResolutionImpossible y **la imagen no se puede construir**. Pasa por
# alto porque nadie la había construido.
#
# Es además la misma política de Admon (requirements.txt, 2026-09-27): las
# transitivas entran solas donde toque y no se fijan. En producción corre la 50.
cryptography>=49
